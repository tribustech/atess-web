```ts
// src/components/home/FlooringSystem/flooring-systems.ts
export type FlooringLayerTextures = {
  map?: string;          // diffuse / albedo (sRGB)
  normalMap?: string;    // tangent-space normal (linear)
  roughnessMap?: string; // roughness (linear)
};

export type FlooringLayer = {
  id: string;            // unique within a system, kebab-case
  label: string;         // user-facing Romanian, e.g. "Amorsă (primer)"
  description: string;   // one-sentence Romanian, professional register
  size: [number, number];   // x, z footprint (scene units)
  thickness: number;        // box height (scene units; proportional, not metric)
  explodeOffset: number;    // additive Y at progress=1 (0 for the base)
  textures?: FlooringLayerTextures;
  tint?: string;            // hex; used when no diffuse map
  tiling?: [number, number];
  roughness?: number;       // default 0.9
  normalScale?: number;     // default 1
};

export type FlooringSystemId =
  | 'spray-sportiv'
  | 'loc-de-joaca'
  | 'teren-sport-outdoor'
  | 'piatra-resin'
  | 'gazon-sintetic';

export type FlooringSystem = {
  id: FlooringSystemId;
  /** Category slug this system illustrates (matches systems.json category axis). */
  category: string;
  /** User-facing system name, Romanian. */
  name: string;
  /** Eyebrow above the heading, e.g. "Construcția sistemului". */
  eyebrow: string;
  /** Intro paragraph under the heading, Romanian, professional register. */
  intro: string;
  /** Ordered bottom -> top. Layer count is DERIVED from this array — never hard-coded. */
  layers: FlooringLayer[];
};

export const FLOORING_SYSTEMS: Record<FlooringSystemId, FlooringSystem>;
export const FLOORING_SYSTEM_LIST: FlooringSystem[];
export function getFlooringSystem(id: FlooringSystemId): FlooringSystem;
export function getFlooringSystemByCategory(category: string): FlooringSystem | undefined;
export const DEFAULT_FLOORING_SYSTEM_ID: FlooringSystemId; // 'spray-sportiv'
```

```ts
export const FLOORING_SYSTEMS: Record<FlooringSystemId, FlooringSystem> = {
  'spray-sportiv': SPRAY_SPORTIV,

export const FLOORING_SYSTEM_LIST: FlooringSystem[] = Object.values(FLOORING_SYSTEMS);

export const DEFAULT_FLOORING_SYSTEM_ID: FlooringSystemId = 'spray-sportiv';

export function getFlooringSystem(id: FlooringSystemId): FlooringSystem {
  return FLOORING_SYSTEMS[id];
}

export function getFlooringSystemByCategory(category: string): FlooringSystem | undefined {
  return FLOORING_SYSTEM_LIST.find((s) => s.category === category);
}

/** Collapsed-state Y center of a layer inside a system, given cumulative thickness below. */
```

```ts
  RepeatWrapping,
  SRGBColorSpace,
  type Texture,

import type { FlooringSystem } from './flooring-systems';

function textureUrlsFor(system: FlooringSystem): string[] {
  return Array.from(
    new Set(
      system.layers
        .flatMap((l) => [l.textures?.map, l.textures?.normalMap, l.textures?.roughnessMap])
        .filter((v): v is string => typeof v === 'string')
    )
  );
}

function configureTexture(tex: Texture, tiling: [number, number], isColor: boolean, aniso: number) {
  tex.wrapS = RepeatWrapping;
  tex.wrapT = RepeatWrapping;
  tex.repeat.set(tiling[0], tiling[1]);
  tex.anisotropy = aniso;
  if (isColor) tex.colorSpace = SRGBColorSpace;
  tex.needsUpdate = true;
}

export function useLayerMaterials(
  system: FlooringSystem
): Record<string, MeshStandardMaterial> {
  const urls = useMemo(() => textureUrlsFor(system), [system]);
  const textures = useTexture(urls) as Texture[];
  const gl = useThree((s) => s.gl);

  return useMemo(() => {
    const byUrl = new Map<string, Texture>();
    urls.forEach((url, i) => byUrl.set(url, textures[i]));

    const aniso = Math.min(gl.capabilities.getMaxAnisotropy(), 8);
    const out: Record<string, MeshStandardMaterial> = {};

    for (const layer of system.layers) {
      const mat = new MeshStandardMaterial({
        roughness: layer.roughness ?? 0.9,
        metalness: 0,
      });

      if (layer.tint) mat.color = new Color(layer.tint);

      const tiling = layer.tiling ?? [1, 1];

      const mapUrl = layer.textures?.map;
      if (mapUrl) {
        const tex = byUrl.get(mapUrl)!.clone();
        configureTexture(tex, tiling, true, aniso);
        mat.map = tex;
      }

      const normalUrl = layer.textures?.normalMap;
      if (normalUrl) {
        const tex = byUrl.get(normalUrl)!.clone();
        configureTexture(tex, tiling, false, aniso);
        mat.normalMap = tex;
        mat.normalScale.set(layer.normalScale ?? 1, layer.normalScale ?? 1);
      }

      const roughUrl = layer.textures?.roughnessMap;
      if (roughUrl) {
        const tex = byUrl.get(roughUrl)!.clone();
        configureTexture(tex, tiling, false, aniso);
        mat.roughnessMap = tex;
      }

      mat.needsUpdate = true;
      out[layer.id] = mat;
    }

    return out;
  }, [textures, gl, system, urls]);
}
```

```ts
  return 1 - Math.pow(1 - t, 3);
}

export function LayerStack({

  progressRef,
  hoveredIdRef,
  onHoverChange,
  autoRotate,
  interactive,
```
